## เปรียบเทียบโครงสร้าง

โปรเจกต์ TypeScript/Node มี `package.json`, build tool, โฟลเดอร์ `dist/` และ `node_modules` ส่วนโปรเจกต์ Go เรียบง่ายกว่า: `go.mod` ประกาศ module และ dependency ไม่มีโฟลเดอร์ output แยกต่างหาก — `go build` สร้าง binary ไฟล์เดียว

export const tsLayout = `// Express / NestJS project tree
my-api/
  package.json
  tsconfig.json
  src/
    main.ts
    app.module.ts          // NestJS root module
    books/
      books.controller.ts
      books.service.ts
      books.dto.ts
    middleware/
      logger.middleware.ts
  node_modules/`;

export const goLayout = `// Echo project tree
books-api/
  go.mod
  go.sum
  main.go                  // entry point; wire everything here
  internal/
    handler/
      books.go             // HTTP handlers (parse, validate, call service)
    service/
      books.go             // business logic
    store/
      books.go             // database queries
  model/
    book.go                // shared structs / domain types`;

## go.mod — package.json ของ Go

`go.mod` ประกาศ module path (ใช้เป็น prefix ของ import) และ Go version ขั้นต่ำ `go get` เพิ่ม dependency และ `go mod tidy` ลบอันที่ไม่ได้ใช้

export const tsPackageJson = `// package.json
{
  "name": "books-api",
  "version": "1.0.0",
  "dependencies": {
    "express": "^4.18.2"
  },
  "devDependencies": {
    "typescript": "^5.0.0"
  }
}`;

export const goModFile = `// go.mod
module github.com/you/books-api

go 1.27

require (
    github.com/labstack/echo/v5 v5.3.1
)`;

## main.go — เชื่อมแอปพลิเคชัน

ใน NestJS `main.ts` เรียก `NestFactory.create` ส่วนใน Echo `main.go` เรียก `echo.New()` ลงทะเบียน middleware กับ route แล้วเรียก `e.Start` ทุกอย่างเป็น Go code ที่ชัดเจน — ไม่มี decorator ไม่มี DI container

export const tsMainTs = `// NestJS main.ts
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  app.setGlobalPrefix('api');
  await app.listen(3000);
}
bootstrap();`;

export const goMainGo = `// main.go
package main

import (
    "log/slog"
    "os"

    "github.com/labstack/echo/v5"
    "github.com/labstack/echo/v5/middleware"

    "github.com/you/books-api/internal/handler"
    "github.com/you/books-api/internal/store"
)

func main() {
    e := echo.New()

    // Middleware
    e.Use(middleware.RequestLogger())
    e.Use(middleware.Recover())

    // Dependency wiring (manual, no container)
    bookStore := store.NewBookStore()
    bookHandler := handler.NewBookHandler(bookStore)

    // Routes
    api := e.Group("/api")
    api.GET("/books", bookHandler.List)
    api.POST("/books", bookHandler.Create)
    api.GET("/books/:id", bookHandler.Get)
    api.PUT("/books/:id", bookHandler.Update)
    api.DELETE("/books/:id", bookHandler.Delete)

    if err := e.Start(":1323"); err != nil {
        slog.Error("server stopped", "err", err)
        os.Exit(1)
    }
}`;

<TsGo ts={tsMainTs} go={goMainGo} />

`e.Start` จะ block จนกว่า server จะหยุด แล้วคืนสาเหตุมาเป็น error — `*slog.Logger` ไม่มี `Fatal` คุณจึงต้อง log และ exit เอง ถ้าต้องการ graceful shutdown หรือ TLS ให้ใช้ `echo.StartConfig` แทน:

```go
sc := echo.StartConfig{Address: ":1323", GracefulTimeout: 10 * time.Second}
if err := sc.Start(ctx, e); err != nil {   // returns when ctx is cancelled
    slog.Error("server stopped", "err", err)
}
```

<GoOnly title="ทำไมต้องใช้ internal/?">

โฟลเดอร์ `internal/` เป็น feature ของภาษา Go: package ที่อยู่ข้างในจะถูก import ได้จากโค้ดภายใน module เดียวกันเท่านั้น นี่คือวิธี idiomatic ที่ทำให้ handler, service และ store เป็น private — เทียบเท่ากับการ mark NestJS module เป็น non-exported ถ้าพยายาม import `github.com/you/books-api/internal/handler` จากนอก module จะ error ตอน compile

</GoOnly>
